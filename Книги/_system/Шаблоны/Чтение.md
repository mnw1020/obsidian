<%*
const norm = (v) => String(v ?? "").replace(/[\u0000-\u001F\u007F]/g, " ").replace(/\s+/g, " ").trim();
const safeName = (v, max = 210) => {
  let s = norm(v).replace(/[<>:"/\\|?*]/g, "-").replace(/\s+/g, " ").replace(/[. ]+$/g, "") || "Без названия";
  while (new TextEncoder().encode(s).length > max && s.length > 1) s = s.slice(0, -1).trim();
  return s || "Без названия";
};
const current = tp.config.target_file;

let existing = {};
await tp.app.fileManager.processFrontMatter(current, (fm) => {
  existing = JSON.parse(JSON.stringify(fm));
});

let bookValue = norm(existing.book);
if (!bookValue) {
  new tp.obsidian.Notice("Создавай перечитывание через + New в разделе История чтений внутри книги");
  return;
}

const m = bookValue.match(/^\[\[([^\]|]+)(?:\|[^\]]+)?\]\]$/);
const bookTarget = m ? m[1] : bookValue;
const bookFile = tp.app.metadataCache.getFirstLinkpathDest(bookTarget, current.path);
if (!bookFile) {
  new tp.obsidian.Notice("Не удалось найти книгу: " + bookTarget);
  return;
}

const readDate = norm(await tp.system.prompt("Дата чтения (YYYY-MM-DD или YYYY-MM)", tp.date.now("YYYY-MM-DD")));
if (!/^\d{4}-\d{2}(?:-\d{2})?$/.test(readDate)) {
  new tp.obsidian.Notice("Дата должна быть в формате YYYY-MM-DD или YYYY-MM");
  return;
}

let rating = null;
const ratingRaw = norm(await tp.system.prompt("Новая оценка 1-10 (можно оставить пустой)", ""));
if (ratingRaw) {
  const parsed = Number(ratingRaw);
  if (!Number.isInteger(parsed) || parsed < 1 || parsed > 10) {
    new tp.obsidian.Notice("Оценка должна быть целым числом от 1 до 10");
    return;
  }
  rating = parsed;
}

await tp.app.fileManager.processFrontMatter(current, (fm) => {
  fm.kind = "reading";
  fm.book = `[[${bookFile.path.replace(/\.md$/, "")}]]`;
  fm.read_date = readDate;
  fm.rating = rating;
});

await tp.app.fileManager.processFrontMatter(bookFile, (fm) => {
  fm.date = readDate;
  if (rating !== null) fm.rating = rating;
});

const cache = tp.app.metadataCache.getFileCache(bookFile);
const title = norm(cache?.frontmatter?.title) || bookFile.basename;
const authorValues = cache?.frontmatter?.authors;
const firstAuthorLink = Array.isArray(authorValues) ? authorValues[0] : authorValues;
let author = norm(firstAuthorLink) || "Не указан";
const am = author.match(/^\[\[[^\]|]+(?:\|([^\]]+))?\]\]$/);
if (am && am[1]) author = am[1];

const base = safeName(`${readDate} - ${author} - ${title}`);
let finalName = base;
let n = 2;
while (true) {
  const path = `Книги/Чтения/${finalName}.md`;
  const found = tp.app.vault.getAbstractFileByPath(path);
  if (!found || found.path === current.path) break;
  finalName = safeName(`${base} (${n++})`, 220);
}
if (current.path !== `Книги/Чтения/${finalName}.md`) {
  await tp.file.move(`Книги/Чтения/${finalName}`);
}
%>
