<%*
const norm = (v) => String(v ?? "").replace(/[\u0000-\u001F\u007F]/g, " ").replace(/\s+/g, " ").trim();
const safeName = (v) => norm(v).replace(/[<>:"/\\|?*]/g, "-").replace(/\s+/g, " ").replace(/[. ]+$/g, "").slice(0, 170) || "Без названия";
const stripLink = (v) => {
  const s = norm(v);
  const m = s.match(/^\[\[([^\]|]+)(?:\|[^\]]+)?\]\]$/);
  return m ? m[1] : s;
};
const current = tp.config.target_file;
let existing = {};
await tp.app.fileManager.processFrontMatter(current, (fm) => { existing = JSON.parse(JSON.stringify(fm)); });
let title = norm(existing.title);
if (!title) title = norm(await tp.system.prompt("Название книги"));
if (!title) {
  new tp.obsidian.Notice("Название книги обязательно");
  return;
}
let authorDefault = Array.isArray(existing.authors) ? existing.authors.join(", ") : norm(existing.authors);
let authorRaw = norm(await tp.system.prompt("Автор(ы), через запятую", authorDefault));
let authors = authorRaw.split(",").map(norm).filter(Boolean);
if (!authors.length) authors = ["Не указан"];
let category = norm(existing.category);
if (!["fiction", "nonfiction"].includes(category)) {
  category = await tp.system.suggester(["Художественная", "Non-fiction"], ["fiction", "nonfiction"], false, "Тип книги");
}
if (!category) category = "fiction";
let status = norm(existing.status);
if (!["want", "reading", "read", "dropped"].includes(status)) {
  status = await tp.system.suggester(["Хочу прочитать", "Читаю", "Прочитано", "Брошено"], ["want", "reading", "read", "dropped"], false, "Статус");
}
if (!status) status = "want";
let existingSeries = stripLink(existing.series);
let seriesName = existingSeries ? existingSeries.split("/").pop() : norm(await tp.system.prompt("Серия (необязательно)", ""));
let seriesIndex = existing.series_index ?? null;
if (seriesName && !seriesIndex) {
  const rawIndex = norm(await tp.system.prompt("Номер в серии (необязательно)", ""));
  const parsed = Number(rawIndex);
  if (Number.isFinite(parsed) && parsed > 0) seriesIndex = parsed;
}
let seriesLink = null;
if (seriesName) {
  const seriesFileName = safeName(seriesName);
  const seriesPath = `Книги/Серии/${seriesFileName}.md`;
  seriesLink = `[[Книги/Серии/${seriesFileName}]]`;
  if (!tp.app.vault.getAbstractFileByPath(seriesPath)) {
    const seriesContent = `---\nkind: book_series\ntitle: ${JSON.stringify(seriesName)}\nauthors:\n  - ${JSON.stringify(authors[0])}\n---\n\n# ${seriesName}\n\n## Книги серии\n\n![[Книги/Книги.base#Книги серии]]\n`;
    await tp.app.vault.create(seriesPath, seriesContent);
  }
}
await tp.app.fileManager.processFrontMatter(current, (fm) => {
  fm.kind = "book";
  fm.title = title;
  fm.authors = authors;
  fm.category = category;
  fm.status = status;
  if (fm.rating === undefined) fm.rating = null;
  if (fm.last_read === undefined) fm.last_read = null;
  if (fm.read_count === undefined) fm.read_count = 0;
  fm.series = seriesLink;
  fm.series_index = seriesIndex || null;
});
const base = safeName(`${authors[0]} - ${title}`);
let finalName = base;
let n = 2;
while (true) {
  const path = `Книги/Карточки/${finalName}.md`;
  const found = tp.app.vault.getAbstractFileByPath(path);
  if (!found || found.path === current.path) break;
  finalName = `${base} (${n++})`;
}
if (current.path !== `Книги/Карточки/${finalName}.md`) {
  await tp.file.move(`Книги/Карточки/${finalName}`);
}
%>
