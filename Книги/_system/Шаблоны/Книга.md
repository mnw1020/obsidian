<%*
const norm = (v) => String(v ?? "").replace(/[\u0000-\u001F\u007F]/g, " ").replace(/\s+/g, " ").trim();
const safeName = (v, max = 170) => {
  let s = norm(v).replace(/[<>:"/\\|?*]/g, "-").replace(/\s+/g, " ").replace(/[. ]+$/g, "") || "Без названия";
  while (new TextEncoder().encode(s).length > max && s.length > 1) s = s.slice(0, -1).trim();
  return s || "Без названия";
};
const stripLink = (v) => {
  const s = norm(v);
  const m = s.match(/^\[\[([^\]|]+)(?:\|[^\]]+)?\]\]$/);
  return m ? m[1] : s;
};
const yamlText = (s) => JSON.stringify(norm(s));

const current = tp.config.target_file;
const title = norm(await tp.system.prompt("Название книги"));
if (!title) {
  new tp.obsidian.Notice("Название книги обязательно");
  return;
}

const authorRaw = norm(await tp.system.prompt("Автор(ы), через запятую"));
let authors = authorRaw.split(",").map(norm).filter(Boolean);
if (!authors.length) authors = ["Не указан"];

const category = await tp.system.suggester(
  ["Художественная", "Non-fiction"],
  ["Художественные", "Non-fiction"],
  false,
  "Раздел"
);
if (!category) return;

const series = norm(await tp.system.prompt("Серия (необязательно)", ""));
let seriesIndex = null;
if (series) {
  const rawIndex = norm(await tp.system.prompt("Номер в серии (необязательно)", ""));
  if (rawIndex) {
    const parsed = Number(rawIndex);
    if (!Number.isInteger(parsed) || parsed < 1) {
      new tp.obsidian.Notice("Номер серии должен быть положительным целым числом");
      return;
    }
    seriesIndex = parsed;
  }
}

const readDate = norm(await tp.system.prompt("Дата чтения YYYY-MM-DD или YYYY-MM (пусто, если не читал)", ""));
if (readDate && !/^\d{4}-\d{2}(?:-\d{2})?$/.test(readDate)) {
  new tp.obsidian.Notice("Дата должна быть в формате YYYY-MM-DD или YYYY-MM");
  return;
}

let rating = null;
const ratingRaw = norm(await tp.system.prompt("Оценка 1-10 (можно оставить пустой)", ""));
if (ratingRaw) {
  const parsed = Number(ratingRaw);
  if (!Number.isInteger(parsed) || parsed < 1 || parsed > 10) {
    new tp.obsidian.Notice("Оценка должна быть целым числом от 1 до 10");
    return;
  }
  rating = parsed;
}

// Ensure author pages exist.
const authorLinks = [];
for (const author of authors) {
  const authorFile = safeName(author);
  const authorPath = `Книги/Авторы/${authorFile}.md`;
  const authorTarget = authorPath.replace(/\.md$/, "");
  authorLinks.push(`[[${authorTarget}|${author}]]`);
  if (!tp.app.vault.getAbstractFileByPath(authorPath)) {
    const content =
`---
kind: book_author
name: ${yamlText(author)}
---
# ${author}

![[Книги/Книги.base#Автор]]
`;
    await tp.app.vault.create(authorPath, content);
  }
}

// Preserve the current folder convention for an existing author.
const categoryFolder = `Книги/${category}`;
const primaryAuthorTarget = `Книги/Авторы/${safeName(authors[0])}`;
let targetFolder = categoryFolder;
for (const f of tp.app.vault.getMarkdownFiles()) {
  const cache = tp.app.metadataCache.getFileCache(f);
  const fm = cache?.frontmatter;
  if (!fm || fm.kind !== "book" || !f.path.startsWith(categoryFolder + "/")) continue;
  const vals = Array.isArray(fm.authors) ? fm.authors : [fm.authors];
  if (vals.some(v => stripLink(v) === primaryAuthorTarget)) {
    targetFolder = f.parent?.path || categoryFolder;
    break;
  }
}

// File naming follows the preserved folder layout.
let baseName;
if (targetFolder !== categoryFolder) {
  if (series && seriesIndex) baseName = `${series}. ${String(seriesIndex).padStart(2, "0")}. ${title}`;
  else baseName = title;
} else {
  baseName = `${authors[0]}. ${title}`;
}
baseName = safeName(baseName);

let finalName = baseName;
let n = 2;
while (true) {
  const path = `${targetFolder}/${finalName}.md`;
  const found = tp.app.vault.getAbstractFileByPath(path);
  if (!found || found.path === current.path) break;
  finalName = safeName(`${baseName} (${n++})`);
}

await tp.app.fileManager.processFrontMatter(current, (fm) => {
  fm.kind = "book";
  fm.title = title;
  fm.authors = authorLinks;
  fm.date = readDate || null;
  fm.rating = rating;
  fm.series = series || null;
  fm.series_index = seriesIndex;
});

const finalTarget = `${targetFolder}/${finalName}`;
if (current.path !== `${finalTarget}.md`) {
  await tp.file.move(finalTarget);
}

// If the book is already read, create its first reading record automatically.
if (readDate || rating !== null) {
  const bookPath = `${finalTarget}`;
  const datePart = readDate || "без даты";
  let readingBase = safeName(`${datePart} - ${authors[0]} - ${title}`, 210);
  let readingName = readingBase;
  let i = 2;
  while (tp.app.vault.getAbstractFileByPath(`Книги/Чтения/${readingName}.md`)) {
    readingName = safeName(`${readingBase} (${i++})`, 220);
  }
  const readingContent =
`---
kind: reading
book: ${yamlText(`[[${bookPath}]]`)}
read_date: ${yamlText(readDate)}
rating: ${rating ?? ""}
---

## Впечатление

`;
  await tp.app.vault.create(`Книги/Чтения/${readingName}.md`, readingContent);
}
%>
