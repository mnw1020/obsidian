function neighborAggregate(model,rated,ref,mode) {
    const out=new Map();
    if(!model?.users?.length) return out;
    const watched=new Set(rated.filter(x=>x.modelKey).map(x=>x.modelKey));
    const personalMean=mean(rated.map(x=>x.rating).filter(Number.isFinite))||6;
    for(const u of model.users){
        let base=Number(u?.seed_similarity);
        if(!Number.isFinite(base)||base<=0.02) continue;
        const pairs=u.ratings||[]; let refR=null;
        if(ref.modelKey){for(const p of pairs){if(p?.[0]===ref.modelKey){refR=Number(p[1]);break;}}}
        let refFactor=0.48;
        if(refR!==null&&Number.isFinite(refR)&&ref.rating!==null){
            const match=Math.exp(-Math.abs(refR-ref.rating)/2.0);
            if(mode==="similar") refFactor=0.38+0.62*(0.72*match+0.28*clamp((refR-4.5)/5.5,0,1));
            else if(mode==="alternative") refFactor=0.40+0.60*match;
            else refFactor=0.48+0.52*match;
        }
        const w=base*refFactor;if(w<=0.008)continue;
        const um=Number.isFinite(Number(u.mean))?Number(u.mean):(mean(pairs.map(p=>Number(p?.[1])).filter(Number.isFinite))||6);
        for(const p of pairs){
            const key=p?.[0],r=Number(p?.[1]);
            if(!key||!Number.isFinite(r)||watched.has(key)||key===ref.modelKey)continue;
            let a=out.get(key);if(!a){a={num:0,den:0,count:0};out.set(key,a);}
            a.num+=w*(r-um);a.den+=Math.abs(w);a.count++;
        }
    }
    for(const a of out.values()){
        const raw=a.den>0?clamp(personalMean+a.num/a.den,1,10):personalMean;
        const reliability=a.count/(a.count+6);
        a.pred=clamp(personalMean+(raw-personalMean)*reliability,1,10);
        a.support=clamp(Math.log1p(a.count)/Math.log(40),0,1);
    }
    return out;
}

function finalScore(x,ref,mode){
    const p=clamp((x.pred-1)/9,0,1),kp=x.kpSim||0,th=x.localTheme||0,sup=x.support||0,era=eraScore(ref,x);
    const multi=Math.min(0.06,Math.max(0,x.sources.size-1)*0.025);
    let s;
    if(mode==="alternative")s=0.31*kp+0.39*p+0.13*th+0.07*sup+0.10*era+multi;
    else if(mode==="similar")s=0.36*kp+0.31*p+0.14*th+0.07*sup+0.12*era+multi;
    else s=0.33*kp+0.35*p+0.14*th+0.07*sup+0.11*era+multi;

    // поэтому возраст не превращаем в жесткий фильтр.
    if(x.sources?.has("КП похожие"))s+=0.035;
    if(x.count>0&&x.count<MIN_NEIGHBORS&&!x.kpSim)s-=0.08;
    return s;
}

function pickFinal(list,ref,mode){
    let sorted=[...list].sort((a,b)=>b.score-a.score||b.kpSim-a.kpSim||b.pred-a.pred||b.count-a.count);
    if(mode==="alternative"&&ref.rating!==null){
        const cut=Math.max(5.5,ref.rating+0.8);
        sorted=[...sorted.filter(x=>x.pred>=cut),...sorted.filter(x=>x.pred<cut)];
    } else if(mode==="similar"){
        sorted=[...sorted.filter(x=>x.pred>=5.5),...sorted.filter(x=>x.pred<5.5)];
    }

    const ry=Number(ref?.year)||0;
    if(ry<2000)return sorted.slice(0,LIMIT);

    // Для современных фильмов делаем выдачу по эпохам. MovieLens сильно
    // смещен к старым популярным фильмам: у них просто больше оценок.
    // Минимум половину TOP стараемся набрать из фильмов не старше ~12 лет
    // относительно референса, но официальные КП-похожие не отбрасываем.
    const recentCut=Math.max(2000,ry-12);
    const oldCut=ry-25;
    const recent=sorted.filter(x=>(Number(x.year)||0)>=recentCut);
    const rest=sorted.filter(x=>(Number(x.year)||0)<recentCut);
    const out=[],seen=new Set();
    const add=x=>{const k=x.modelKey||x.kpId||`${titleKey(x.enTitle||x.ruTitle)}:${x.year}`;if(seen.has(k)||out.length>=LIMIT)return;seen.add(k);out.push(x);};
    for(const x of recent.slice(0,Math.min(12,LIMIT)))add(x);
    let veryOld=0;
    for(const x of sorted){
        const y=Number(x.year)||0;
        if(y&&y<oldCut&&!x.sources?.has("КП похожие")&&veryOld>=4)continue;
        const before=out.length;add(x);if(out.length>before&&y&&y<oldCut)veryOld++;
        if(out.length>=LIMIT)break;
    }
    for(const x of rest){add(x);if(out.length>=LIMIT)break;}
    return out.slice(0,LIMIT);
}
