```dataviewjs
const runtimeFile = app.vault.getAbstractFileByPath("Кино/_system/recommendations.js");
if (!runtimeFile) {
    dv.paragraph("Не найден скрипт Кино/_system/recommendations.js.");
} else {
    const runtimeSource = await app.vault.read(runtimeFile);
    const runRecommendations = new Function("dv", "app", "obsidian", "require", `return (async () => {${runtimeSource}\n})();`);
    await runRecommendations(dv, app, typeof obsidian !== "undefined" ? obsidian : {}, typeof require === "function" ? require : null);
}
```
