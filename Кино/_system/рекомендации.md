# Рекомендации

> [!info] Как работает
> Нажми **"🔎 Найти похожие"** в карточке. Основная выдача строится сразу из твоих текущих оценок и MovieLens. Интернет IMDb используется только как дополнительный источник и больше не может заблокировать страницу.
>
> - **Оценка 1-4**: ищутся более перспективные альтернативы в близких жанрах.
> - **Оценка 6-10**: ищутся близкие по теме фильмы, которые вкусовые соседи оценивают высоко.
> - **Оценка 5 / нет оценки**: баланс темы и персонального прогноза.
>
> Уже оцененные тобой произведения исключаются. В выдаче - до **20** рекомендаций.

const ROOT = "Кино";
const STATE_PATH = `${ROOT}/_system/Прогноз/рекомендации_state.json`;
const MODEL_PATH = `${ROOT}/_system/Прогноз/movielens_neighbors.json`;
const LIMIT = 20;
const CANDIDATE_POOL = 220;
const clamp = (x,a,b) => Math.max(a, Math.min(b,x));
const num = v => {
    if (v === null || v === undefined || v === "") return null;
    const n = Number(String(v).replace(",",".").replace(/[^0-9+\-.]/g,""));
    return Number.isFinite(n) ? n : null;
};
const arr = v => Array.isArray(v) ? v : (v === null || v === undefined || v === "" ? [] : [v]);
const mean = xs => xs.length ? xs.reduce((a,b)=>a+b,0)/xs.length : null;
const norm = s => String(s ?? "").normalize("NFC").toLocaleLowerCase("ru").replace(/ё/g,"е").replace(/[^a-zа-я0-9]+/gi," ").trim();
const setCos = (a,b) => {
    if (!a.size || !b.size) return 0;
    let n=0; for (const x of a) if (b.has(x)) n++;
    return n / Math.sqrt(a.size*b.size);
};
const fmt = v => Number.isFinite(v) ? Number(v).toFixed(1) : "-";

function fmOf(file) { return app.metadataCache.getFileCache(file)?.frontmatter || {}; }
function isMedia(file) {
    if (!file || file.extension !== "md" || !file.path.startsWith(ROOT + "/")) return false;
    const rel=file.path.slice(ROOT.length+1); if (rel.includes("/")) return false;
    const tags=arr(fmOf(file).tags).map(x=>String(x).replace(/^#/,""));
    return tags.includes("movies") || tags.includes("serial");
}
function imdbIdOf(fm) { return String(fm?.["imdb Id"] || "").match(/tt\d{7,12}/i)?.[0]?.toLowerCase() || ""; }
function featureFromFile(file) {
    const fm=fmOf(file);
    return {
        file, localPath:file.path, title:file.basename,
        imdbId:imdbIdOf(fm), rating:num(fm?.["Оценка"]), storedPrediction:num(fm?.["Прогноз оценки"]),
        imdbRating:num(fm?.["Оценка Imdb"]), kpRating:num(fm?.["Оценка Кинопоиск"]), kpId:String(fm?.["Кинопоиск ID"]||"").replace(/\D/g,""),
        genres:new Set(arr(fm?.["Жанр"]).map(norm).filter(Boolean)),
        year:(String(fm?.["Релиз"]||"").match(/(?:18|19|20)\d{2}/)||[])[0] || "",
        type:arr(fm?.tags).map(x=>String(x).replace(/^#/,"")).includes("serial") ? "series" : "movie"
    };
}
async function loadJson(path) {
    const f=app.vault.getAbstractFileByPath(path); if(!f) return null;
    try { return JSON.parse(await app.vault.read(f)); } catch(_) { return null; }
}
function pearsonCommon(userMap,pairs) {
    const xs=[],ys=[];
    for(const p of pairs||[]){ const r=Number(p?.[1]); if(userMap.has(p?.[0])&&Number.isFinite(r)){xs.push(userMap.get(p[0]));ys.push(r);} }
    const n=xs.length; if(n<5) return {n,weight:0,corr:0};
    const mx=mean(xs), my=mean(ys); let nu=0,vx=0,vy=0;
    for(let i=0;i<n;i++){const dx=xs[i]-mx,dy=ys[i]-my;nu+=dx*dy;vx+=dx*dx;vy+=dy*dy;}
    const corr=(vx>0&&vy>0)?nu/Math.sqrt(vx*vy):0;
    return {n,corr,weight:corr*(n/(n+12))};
}
function calibration(items) {
    const pts=items.filter(x=>x.rating!==null&&x.imdbRating!==null).map(x=>[x.imdbRating,x.rating]);
    if(pts.length<20) return {a:0,b:1};
    const mx=mean(pts.map(x=>x[0])), my=mean(pts.map(x=>x[1])); let cov=0,vx=0;
    for(const [x,y] of pts){cov+=(x-mx)*(y-my);vx+=(x-mx)*(x-mx);}
    const b=vx>1e-9?cov/vx:1; return {a:my-b*mx,b};
}
function publicPred(rating,cal,globalMean){return rating===null?globalMean:clamp(cal.a+cal.b*rating,1,10);}
function currentNeighborAggregate(model,rated,ref,mode) {
    const out=new Map();
    if(!model?.users?.length) return out;
    const userMap=new Map(rated.filter(x=>x.imdbId&&x.rating!==null).map(x=>[x.imdbId,x.rating]));
    const personalMean=mean([...userMap.values()])||6;
    for(const u of model.users){
        const pairs=u.ratings||[]; const sim=pearsonCommon(userMap,pairs);
        if(sim.weight<=0.02) continue;
        let refR=null;
        if(ref.imdbId){ for(const p of pairs){ if(p[0]===ref.imdbId){refR=Number(p[1]);break;} } }
        let refFactor=0.42;
        if(refR!==null && Number.isFinite(refR) && ref.rating!==null){
            const match=Math.exp(-Math.abs(refR-ref.rating)/2.0);
            if(mode==="similar") refFactor=0.35+0.65*(0.70*match+0.30*clamp((refR-4.5)/5.5,0,1));
            else if(mode==="alternative") refFactor=0.35+0.65*match;
            else refFactor=0.45+0.55*match;
        }
        const w=sim.weight*refFactor; if(w<=0.008) continue;
        const um=Number.isFinite(Number(u.mean))?Number(u.mean):(mean(pairs.map(p=>Number(p[1])).filter(Number.isFinite))||6);
        for(const p of pairs){
            const id=p[0], r=Number(p[1]); if(!/^tt\d+$/i.test(id)||!Number.isFinite(r)||userMap.has(id)||id===ref.imdbId) continue;
            let a=out.get(id); if(!a){a={num:0,den:0,count:0,raw:0,rawDen:0};out.set(id,a);}
            a.num+=w*(r-um); a.den+=Math.abs(w); a.count++; a.raw+=w*r; a.rawDen+=Math.abs(w);
        }
    }
    for(const a of out.values()){
        a.pred=a.den>0?clamp(personalMean+a.num/a.den,1,10):personalMean;
        a.support=clamp(Math.log1p(a.count)/Math.log(100),0,1);
    }
    return out;
}
function modelCandidate(id,a,model,localById,ref,mode,cal,globalMean) {
    const meta=model?.catalog?.[id]||{}; const local=localById.get(id)||null;
    const genres=new Set((local?[...local.genres]:(meta.genres||[]).map(norm)).filter(Boolean));
    const theme=setCos(ref.genres,genres);
    let pred=a?.pred ?? null; let sources=[];
    if(a){sources.push("MovieLens");}
    if(local && local.storedPrediction !== null){ pred=pred===null?local.storedPrediction:0.72*pred+0.28*local.storedPrediction; sources.push("локальный прогноз"); }
    if(pred===null) pred=publicPred(local?.imdbRating??null,cal,globalMean);
    const pnorm=(pred-1)/9;
    let score;
    if(mode==="alternative") score=0.75*pnorm+0.22*theme+0.03*(a?.support||0);
    else if(mode==="similar") score=0.68*pnorm+0.27*theme+0.05*(a?.support||0);
    else score=0.72*pnorm+0.23*theme+0.05*(a?.support||0);
    return {
        id,pred,score,theme,support:a?.support||0,count:a?.count||0,
        title:local?.title||meta.title||`IMDb ${id}`,
        year:local?.year||meta.year||"", genres, local,
        imdbRating:local?.imdbRating??null, kpRating:local?.kpRating??null, kpId:local?.kpId||"",
        sources
    };
}

const http = (typeof requestUrl !== "undefined" && requestUrl) || ((typeof obsidian !== "undefined" && obsidian.requestUrl) ? obsidian.requestUrl : null);
async function withTimeout(p,ms){ return await Promise.race([p,new Promise((_,rej)=>setTimeout(()=>rej(new Error("timeout")),ms))]); }
async function imdbMoreLike(id){
    if(!http||!id) return [];
    const query=`query Recommendations($id: ID!) { title(id:$id) { moreLikeThisTitles(first:40) { edges { node { id titleText { text } releaseYear { year } ratingsSummary { aggregateRating } titleGenres { genres { genre { text } } } } } } } }`;
    try{
        const r=await withTimeout(http({url:"https://graphql.imdb.com/",method:"POST",headers:{"Content-Type":"application/json","User-Agent":"Mozilla/5.0"},body:JSON.stringify({query,variables:{id}})}),4500);
        const j=r?.json; if(!j?.data?.title?.moreLikeThisTitles?.edges) return [];
        return j.data.title.moreLikeThisTitles.edges.map((e,i)=>{const n=e?.node||{};return {id:String(n.id||"").toLowerCase(),rank:i+1,title:n.titleText?.text||"",year:n.releaseYear?.year||"",rating:num(n.ratingsSummary?.aggregateRating),genres:new Set((n.titleGenres?.genres||[]).map(x=>norm(x?.genre?.text)).filter(Boolean))};}).filter(x=>/^tt\d+$/i.test(x.id));
    }catch(_){return [];}
}
async function imdbSuggest(id){
    if(!http||!id) return null;
    try{
        const r=await withTimeout(http({url:`https://v3.sg.media-imdb.com/suggestion/x/${encodeURIComponent(id)}.json`,method:"GET",headers:{"User-Agent":"Mozilla/5.0"}}),3000);
        const d=r?.json?.d||[]; const x=d.find(v=>String(v?.id||"").toLowerCase()===id.toLowerCase())||d[0];
        if(!x) return null; return {title:x.l||"",year:x.y||"",poster:x.i?.imageUrl||""};
    }catch(_){return null;}
}
async function mapLimit(items,limit,fn){const out=new Array(items.length);let next=0;async function w(){while(true){const i=next++;if(i>=items.length)break;try{out[i]=await fn(items[i],i);}catch(_){out[i]=null;}}}await Promise.all(Array.from({length:Math.min(limit,items.length)},()=>w()));return out;}

function render(container,list,refFile,statusText,catalogReady){
    container.empty();
    const status=container.createDiv(); status.style.margin="10px 0 14px"; status.style.padding="9px 11px"; status.style.border="1px solid var(--background-modifier-border)"; status.style.borderRadius="8px"; status.textContent=statusText;
    if(!catalogReady){
        const w=container.createDiv(); w.style.margin="8px 0 14px"; w.style.padding="9px 11px"; w.style.background="var(--background-secondary)"; w.style.borderRadius="8px";
        w.innerHTML="⚠ Для точного фильтра <b>по той же теме</b> один раз запусти <code>Кино/_system/Обновить каталог рекомендаций.bat</code>. Без каталога рекомендации все равно строятся по вкусовым соседям.";
    }
    if(!list.length){container.createEl("p",{text:"Кандидаты не найдены. Проверь, что movielens_neighbors.json существует и содержит users."});return;}

    list.forEach((x,i)=>{
        const row=table.createEl("tr");

        if(x.year)td.appendText(` (${x.year})`);
        const g=td.createEl("div");g.style.fontSize="0.8em";g.style.opacity="0.7";g.textContent=[...x.genres].slice(0,4).join(", ");
        const p=row.createEl("td",{text:fmt(x.pred)});p.style.fontWeight="700";p.style.padding="7px 6px";

        const sd=row.createEl("td",{text:[...new Set(x.sources)].join(" + ")});sd.style.padding="7px 6px";sd.style.fontSize="0.85em";

async function main(){
    const state=await loadJson(STATE_PATH);
    if(!state?.reference){dv.paragraph("Открой карточку фильма и нажми **🔎 Найти похожие**.");return;}
    const refFile=app.vault.getAbstractFileByPath(state.reference);
    if(!refFile||!isMedia(refFile)){dv.paragraph(`Не удалось открыть референс: ${state.reference}`);return;}
    const model=await loadJson(MODEL_PATH);
    if(!model?.users?.length){dv.paragraph("Не найдена готовая MovieLens-модель. Сначала запусти `Кино/_system/Обновить модель MovieLens.bat`.");return;}

    const allFiles=app.vault.getMarkdownFiles().filter(isMedia); const items=allFiles.map(featureFromFile); const ref=items.find(x=>x.file.path===refFile.path);
    const rated=items.filter(x=>x.file.path!==refFile.path&&x.rating!==null&&x.rating>=1&&x.rating<=10);
    const localById=new Map(items.filter(x=>x.imdbId).map(x=>[x.imdbId,x]));
    const watchedIds=new Set(items.filter(x=>x.rating!==null&&x.imdbId).map(x=>x.imdbId));
    const mode=ref.rating!==null&&ref.rating<5?"alternative":ref.rating!==null&&ref.rating>5?"similar":"neutral";
    const modeText=mode==="alternative"?"Альтернативы по той же теме":mode==="similar"?"Похожие, которые должны понравиться":"Нейтральный поиск по теме";

    const head=dv.container.createDiv(); head.createEl("h2",{text:`${modeText}: ${refFile.basename}`});
    const hp=head.createEl("p");hp.innerHTML=`Твоя оценка: <b>${ref.rating===null?"нет":fmt(ref.rating)}</b> · Прогноз карточки: <b>${fmt(ref.storedPrediction)}</b>`;
    const result=dv.container.createDiv();

    const cal=calibration(items), globalMean=mean(rated.map(x=>x.rating))||6;
    const agg=currentNeighborAggregate(model,rated,ref,mode);
    let list=[];
    for(const [id,a] of agg){
        if(watchedIds.has(id))continue;
        const x=modelCandidate(id,a,model,localById,ref,mode,cal,globalMean);
        const catalogReady=!!model.catalog && Object.keys(model.catalog).length>0;
        if(catalogReady && ref.genres.size && x.genres.size && x.theme<0.34)continue;
        if(mode==="alternative"&&ref.rating!==null&&x.pred<Math.max(5.5,ref.rating+0.8))continue;
        list.push(x);
    }
    list.sort((a,b)=>b.score-a.score||b.pred-a.pred); list=list.slice(0,CANDIDATE_POOL);
    const catalogReady=!!model.catalog && Object.keys(model.catalog).length>0;
    let top=list.slice(0,LIMIT);
    render(result,top,refFile,`MovieLens: готово · кандидатов после фильтра: ${list.length} · интернет IMDb: проверяется...`,catalogReady);

    // Resolve human-readable titles for old models that do not yet contain catalog metadata.
    const needNames=top.filter(x=>x.title===`IMDb ${x.id}`);
    if(needNames.length){
        const info=await mapLimit(needNames,6,x=>imdbSuggest(x.id));
        needNames.forEach((x,i)=>{const d=info[i];if(d?.title){x.title=d.title;x.year=x.year||d.year;x.sources.push("IMDb");}});
        render(result,top,refFile,`MovieLens: готово · кандидатов после фильтра: ${list.length} · интернет IMDb: названия запрошены`,catalogReady);
    }

    // Optional online signal. Hard timeout means it can never leave the page hanging.
    const online=await imdbMoreLike(ref.imdbId);
    if(online.length){
        const rank=new Map(online.map(x=>[x.id,x]));
        const byId=new Map(list.map(x=>[x.id,x]));
        for(const o of online){
            if(watchedIds.has(o.id)||o.id===ref.imdbId)continue;
            let x=byId.get(o.id);
            if(!x){
                const a=agg.get(o.id)||null; x=modelCandidate(o.id,a,model,localById,ref,mode,cal,globalMean); byId.set(o.id,x); list.push(x);
            }
            x.imdbRank=o.rank; x.sources.push("IMDb похожие");
            if((!x.title||x.title===`IMDb ${x.id}`)&&o.title)x.title=o.title;
            if(!x.year&&o.year)x.year=o.year;
            if(!x.genres.size&&o.genres.size)x.genres=o.genres;
            if(x.imdbRating===null&&o.rating!==null)x.imdbRating=o.rating;
            x.theme=Math.max(x.theme,setCos(ref.genres,x.genres));
            x.score+=0.12*Math.exp(-(o.rank-1)/12)+0.08*x.theme;
        }
        list=list.filter(x=>!catalogReady||!ref.genres.size||!x.genres.size||x.theme>=0.34||x.imdbRank<=20);
        list.sort((a,b)=>b.score-a.score||b.pred-a.pred); top=list.slice(0,LIMIT);
        render(result,top,refFile,`MovieLens: готово · IMDb More Like This: ${online.length} кандидатов · показано: ${top.length}`,catalogReady);
    } else {
        render(result,top,refFile,`MovieLens: готово · IMDb сейчас недоступен или не ответил за 4.5 с · показано: ${top.length}`,catalogReady);
    }
}

await main();
