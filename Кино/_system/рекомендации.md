# Рекомендации

> [!info] Как работает
> Нажми **"🔎 Найти похожие"** в карточке. Основа рекомендаций теперь только **MovieLens + Кинопоиск**. IMDb не запрашивается, не используется как интернет-источник и не показывается в таблице.
>
> - **Оценка 1-4**: ищутся более перспективные альтернативы в близкой теме.
> - **Оценка 6-10**: ищутся близкие по теме фильмы, которые должны тебе понравиться.
> - **Оценка 5 / нет оценки**: баланс темы и персонального прогноза.
>
> Уже оцененные тобой произведения исключаются. В выдаче - до **20** рекомендаций.

```dataviewjs
const ROOT = "Кино";
const STATE_PATH = `${ROOT}/_system/Прогноз/рекомендации_state.json`;
const MODEL_PATH = `${ROOT}/_system/Прогноз/movielens_neighbors.json`;
const KP_CACHE_PATH = `${ROOT}/_system/Прогноз/recommendations_kp_cache.json`;
const LIMIT = 20;
const CANDIDATE_POOL = 220;
const KP_TIMEOUT_MS = 8000;
const KP_GAP_MS = 850;
const MIN_NEIGHBORS = 3;
const MISS_RETRY_MS = 24 * 3600 * 1000;
const KP_CACHE_VERSION = 3;

const clamp = (x,a,b) => Math.max(a, Math.min(b,x));
const num = v => {
    if (v === null || v === undefined || v === "") return null;
    const n = Number(String(v).replace(",",".").replace(/[^0-9+\-.]/g,""));
    return Number.isFinite(n) ? n : null;
};
const arr = v => Array.isArray(v) ? v : (v === null || v === undefined || v === "" ? [] : [v]);
const mean = xs => xs.length ? xs.reduce((a,b)=>a+b,0)/xs.length : null;
const norm = s => String(s ?? "").normalize("NFC").toLocaleLowerCase("ru").replace(/ё/g,"е").replace(/[^a-zа-я0-9]+/gi," ").trim();
const setCos = (a,b) => {
    if (!a.size || !b.size) return 0;
    let n=0; for (const x of a) if (b.has(x)) n++;
    return n / Math.sqrt(a.size*b.size);
};
const fmt = v => Number.isFinite(v) ? Number(v).toFixed(1) : "-";

function titleKey(s) {
    let t=String(s??"").trim().replace(/\s*\((?:18|19|20)\d{2}\)\s*$/," ").trim();
    t=t.replace(/^(.+),\s*(the|a|an)$/i,"$2 $1");
    t=norm(t).replace(/^(the|a|an)\s+/,"");
    return t;
}
function fmOf(file) { return app.metadataCache.getFileCache(file)?.frontmatter || {}; }
function isMedia(file) {
    if (!file || file.extension !== "md" || !file.path.startsWith(ROOT + "/")) return false;
    const rel=file.path.slice(ROOT.length+1); if (rel.includes("/")) return false;
    const tags=arr(fmOf(file).tags).map(x=>String(x).replace(/^#/,""));
    return tags.includes("movies") || tags.includes("serial");
}
function featureFromFile(file) {
    const fm=fmOf(file);
    return {
        file, localPath:file.path,
        ruTitle:file.basename,
        enTitle:String(fm?.["Название"] || file.basename),
        rating:num(fm?.["Оценка"]),
        storedPrediction:num(fm?.["Прогноз оценки"]),
        kpRating:num(fm?.["Оценка Кинопоиск"]),
        kpId:String(fm?.["Кинопоиск ID"]||"").replace(/\D/g,""),
        genres:new Set(arr(fm?.["Жанр"]).map(norm).filter(Boolean)),
        year:(String(fm?.["Релиз"]||"").match(/(?:18|19|20)\d{2}/)||[])[0] || "",
        type:arr(fm?.tags).map(x=>String(x).replace(/^#/,"")).includes("serial") ? "series" : "movie",
        modelKey:""
    };
}
async function loadJson(path) {
    const f=app.vault.getAbstractFileByPath(path); if(!f) return null;
    try { return JSON.parse(await app.vault.read(f)); } catch(_) { return null; }
}
async function saveJson(path,obj) {
    const text=JSON.stringify(obj,null,2);
    const f=app.vault.getAbstractFileByPath(path);
    try {
        if(f) await app.vault.modify(f,text);
        else {
            const folder=path.split("/").slice(0,-1).join("/");
            if(!app.vault.getAbstractFileByPath(folder)) await app.vault.createFolder(folder);
            await app.vault.create(path,text);
        }
    } catch(_) {}
}
function calibration(items) {
    const pts=items.filter(x=>x.rating!==null&&x.kpRating!==null).map(x=>[x.kpRating,x.rating]);
    if(pts.length<20) return {a:0,b:1};
    const mx=mean(pts.map(x=>x[0])), my=mean(pts.map(x=>x[1])); let cov=0,vx=0;
    for(const [x,y] of pts){cov+=(x-mx)*(y-my);vx+=(x-mx)*(x-mx);}
    const b=vx>1e-9?cov/vx:1; return {a:my-b*mx,b};
}
function publicPred(rating,cal,globalMean){return rating===null?globalMean:clamp(cal.a+cal.b*rating,1,10);}

function buildCatalogIndex(model) {
    const byTitle=new Map();
    for(const [key,meta] of Object.entries(model?.catalog||{})) {
        const tk=titleKey(meta?.title||"");
        if(!tk) continue;
        const row={key,year:Number(meta?.year)||0,genres:new Set(arr(meta?.genres).map(norm).filter(Boolean)),title:String(meta?.title||"")};
        if(!byTitle.has(tk)) byTitle.set(tk,[]);
        byTitle.get(tk).push(row);
    }
    return byTitle;
}
function assignModelKeys(items,model) {
    const idx=buildCatalogIndex(model);
    let matched=0;
    for(const x of items) {
        const candidates=idx.get(titleKey(x.enTitle))||[];
        if(!candidates.length) continue;
        const yr=Number(x.year)||0;
        let best=null,bestScore=-1e9;
        for(const c of candidates) {
            const yearDiff=(yr&&c.year)?Math.abs(c.year-yr):0;
            if(yr&&c.year&&yearDiff>1) continue;
            const theme=setCos(x.genres,c.genres);
            const score=(yearDiff===0?5:yearDiff===1?2:0)+theme;
            if(score>bestScore){bestScore=score;best=c;}
        }
        if(!best && candidates.length===1) best=candidates[0];
        if(best){x.modelKey=best.key;matched++;}
    }
    return matched;
}

function neighborAggregate(model,rated,ref,mode) {
    const out=new Map();
    if(!model?.users?.length) return out;
    const watched=new Set(rated.filter(x=>x.modelKey).map(x=>x.modelKey));
    const personalMean=mean(rated.map(x=>x.rating).filter(Number.isFinite))||6;

    for(const u of model.users){
        let base=Number(u?.seed_similarity);
        if(!Number.isFinite(base)||base<=0.02) continue;
        const pairs=u.ratings||[];
        let refR=null;
        if(ref.modelKey){
            for(const p of pairs){ if(p?.[0]===ref.modelKey){refR=Number(p[1]);break;} }
        }
        let refFactor=0.48;
        if(refR!==null && Number.isFinite(refR) && ref.rating!==null){
            const match=Math.exp(-Math.abs(refR-ref.rating)/2.0);
            if(mode==="similar") refFactor=0.38+0.62*(0.72*match+0.28*clamp((refR-4.5)/5.5,0,1));
            else if(mode==="alternative") refFactor=0.40+0.60*match;
            else refFactor=0.48+0.52*match;
        }
        const w=base*refFactor; if(w<=0.008) continue;
        const um=Number.isFinite(Number(u.mean))?Number(u.mean):(mean(pairs.map(p=>Number(p?.[1])).filter(Number.isFinite))||6);
        for(const p of pairs){
            const key=p?.[0], r=Number(p?.[1]);
            if(!key||!Number.isFinite(r)||watched.has(key)||key===ref.modelKey) continue;
            let a=out.get(key); if(!a){a={num:0,den:0,count:0};out.set(key,a);}
            a.num+=w*(r-um); a.den+=Math.abs(w); a.count++;
        }
    }
    for(const a of out.values()){
        const raw=a.den>0?clamp(personalMean+a.num/a.den,1,10):personalMean;
        const reliability=a.count/(a.count+6);
        a.pred=clamp(personalMean+(raw-personalMean)*reliability,1,10);
        a.support=clamp(Math.log1p(a.count)/Math.log(40),0,1);
    }
    return out;
}
function modelCandidate(key,a,model,localByKey,ref,mode,cal,globalMean) {
    const meta=model?.catalog?.[key]||{}; const local=localByKey.get(key)||null;
    const genres=new Set((local?[...local.genres]:arr(meta.genres).map(norm)).filter(Boolean));
    const theme=setCos(ref.genres,genres);
    let pred=a?.pred ?? null;
    if(local && local.storedPrediction!==null) pred=pred===null?local.storedPrediction:0.80*pred+0.20*local.storedPrediction;
    if(pred===null) pred=publicPred(local?.kpRating??null,cal,globalMean);
    const pnorm=(pred-1)/9;
    const support=a?.support||0;
    let score;
    if(mode==="alternative") score=0.60*pnorm+0.25*theme+0.15*support;
    else if(mode==="similar") score=0.56*pnorm+0.31*theme+0.13*support;
    else score=0.58*pnorm+0.28*theme+0.14*support;
    return {
        modelKey:key,pred,score,theme,support,count:a?.count||0,
        ruTitle:local?.ruTitle||"",
        enTitle:local?.enTitle||String(meta.title||key),
        year:local?.year||String(meta.year||""), genres, local, type:local?.type||"",
        kpId:local?.kpId||"", kpRating:local?.kpRating??null
    };
}

const http=(typeof requestUrl!=="undefined"&&requestUrl)||((typeof obsidian!=="undefined"&&obsidian.requestUrl)?obsidian.requestUrl:null);
async function withTimeout(p,ms){return await Promise.race([p,new Promise((_,rej)=>setTimeout(()=>rej(new Error("timeout")),ms))]);}
const sleep = ms => new Promise(r=>setTimeout(r,ms));
let kpNextRequestAt = 0;

function kpSearchUrl(params={}){
    const u=new URL("https://movie-planner.ru/api/public/search");
    for(const [k,v] of Object.entries(params)) if(v!==null&&v!==undefined&&v!=="") u.searchParams.set(k,String(v));
    return u.href;
}
async function kpGetJson(url){
    if(!http) return {ok:false,data:null,status:0,reason:"no-http"};
    for(let attempt=0; attempt<2; attempt++){
        const wait=Math.max(0,kpNextRequestAt-Date.now());
        if(wait) await sleep(wait);
        kpNextRequestAt=Date.now()+KP_GAP_MS;
        try{
            const r=await withTimeout(http({
                url,method:"GET",throw:false,
                headers:{Accept:"application/json","User-Agent":"ObsidianKinoteka/2.5 (personal movie catalogue)"}
            }),KP_TIMEOUT_MS);
            const status=Number(r?.status||0);
            if(status===200) return {ok:true,data:r?.json||null,status};
            if(status===429||status===503){
                const h=Object.entries(r?.headers||{}).find(([k])=>String(k).toLowerCase()==="retry-after")?.[1];
                const sec=Number(h);
                const pause=Number.isFinite(sec)?Math.max(KP_GAP_MS,sec*1000):3500;
                kpNextRequestAt=Math.max(kpNextRequestAt,Date.now()+pause);
                if(attempt===0) continue;
            }
            return {ok:false,data:r?.json||null,status,reason:`http-${status}`};
        }catch(e){
            if(attempt===0){kpNextRequestAt=Math.max(kpNextRequestAt,Date.now()+1600);continue;}
            return {ok:false,data:null,status:0,reason:String(e?.message||e||"request-error")};
        }
    }
    return {ok:false,data:null,status:0,reason:"unknown"};
}
function kpResultFromItem(item,x){
    if(!item) return null;
    const kpId=String(item.kp_id||item.kinopoisk_id||item.id||"").replace(/\D/g,"");
    if(!kpId) return null;
    return {
        v:KP_CACHE_VERSION,found:true,checkedAt:Date.now(),kpId,
        ruTitle:String(item.title||item.title_ru||item.name_ru||item.name||"").trim(),
        enTitle:String(item.title_en||item.original_title||item.name_en||item.alternativeName||"").trim(),
        year:String(item.year||x.year||"").trim(),
        kpRating:num(item.rating_kp??item.kp_rating??item.rating)
    };
}
function candidateScore(item,x){
    const yr=Number(x.year)||0, iy=Number(item?.year)||0;
    if(yr&&iy&&Math.abs(iy-yr)>1) return -999;
    const q=titleKey(x.enTitle);
    const names=[item?.title_en,item?.original_title,item?.name_en,item?.alternativeName,item?.title,item?.name].filter(Boolean).map(titleKey);
    let score=0;
    if(yr&&iy) score += iy===yr?5:2;
    if(names.includes(q)) score+=8;
    else if(names.some(n=>n&&q&&(n.includes(q)||q.includes(n)))) score+=3;
    if(Boolean(item?.is_series)===(x.type==="series")) score+=1;
    return score;
}
async function resolveKinopoisk(x,cache){
    if(x.local?.kpId){
        return {v:KP_CACHE_VERSION,kpId:x.local.kpId,ruTitle:x.local.ruTitle,enTitle:x.local.enTitle,found:true,source:"local",kpRating:x.local.kpRating};
    }
    const old=cache?.[x.modelKey];
    if(old?.found && old?.kpId) return old;
    if(old?.v===KP_CACHE_VERSION && old?.found===false && Date.now()-Number(old.checkedAt||0)<MISS_RETRY_MS) return old;

    const query=`${x.enTitle} ${x.year||""}`.trim();
    const resp=await kpGetJson(kpSearchUrl({q:query,limit:16,type:x.type==="series"?"series":"film",person_limit:0}));
    if(!resp.ok) return {found:false,transient:true,reason:resp.reason||`http-${resp.status||0}`};
    const items=Array.isArray(resp.data?.items)?resp.data.items:[];
    const ranked=items.map(it=>({it,score:candidateScore(it,x)})).filter(z=>z.score>=5).sort((a,b)=>b.score-a.score);
    let chosen=null;
    if(ranked.length && (ranked.length===1 || ranked[0].score>ranked[1].score)) chosen=ranked[0].it;
    else if(ranked.length && ranked[0].score>=12) chosen=ranked[0].it;
    let result=kpResultFromItem(chosen,x);
    if(result){cache[x.modelKey]=result;return result;}
    result={v:KP_CACHE_VERSION,found:false,checkedAt:Date.now(),kpId:"",ruTitle:"",enTitle:"",year:""};
    cache[x.modelKey]=result;
    return result;
}

function makeUi(container){
    const box=container.createDiv(); box.style.margin="10px 0 14px"; box.style.padding="9px 11px"; box.style.border="1px solid var(--background-modifier-border)"; box.style.borderRadius="8px";
    const text=box.createDiv();
    const bar=box.createEl("progress"); bar.max=100; bar.value=0; bar.style.width="100%"; bar.style.marginTop="7px";
    const tableWrap=container.createDiv();
    return {box,text,bar,tableWrap};
}
function setProgress(ui,text,pct){ui.text.textContent=text;ui.bar.value=clamp(pct,0,100);}
function renderTable(ui,list){
    ui.tableWrap.empty();
    if(!list.length){ui.tableWrap.createEl("p",{text:"Кандидаты не найдены."});return;}
    const table=ui.tableWrap.createEl("table"); table.style.width="100%"; table.style.borderCollapse="collapse";
    const hr=table.createEl("tr"); ["Русское название","English","Прогноз"].forEach(h=>{const th=hr.createEl("th",{text:h});th.style.textAlign="left";th.style.padding="7px 8px";});
    for(const x of list){
        const row=table.createEl("tr");
        const ru=row.createEl("td");ru.style.padding="8px";ru.style.minWidth="240px";
        const ruText=x.ruTitle||x.local?.ruTitle||"-";
        if(x.kpId){const a=ru.createEl("a",{text:ruText==="-"?`Кинопоиск ${x.kpId}`:ruText});a.href=`https://www.kinopoisk.ru/film/${x.kpId}/`;a.target="_blank";}else ru.textContent=ruText;
        if(x.year){const s=ru.createEl("span",{text:` (${x.year})`});s.style.opacity="0.7";}

        const en=row.createEl("td");en.style.padding="8px";en.style.minWidth="220px";
        en.textContent=x.enTitle||"-";

        const p=row.createEl("td",{text:fmt(x.pred)});p.style.padding="8px";p.style.fontWeight="700";p.style.fontSize="1.05em";
    }
}
function pickTop(list){
    const strict=list.filter(x=>x.count>=MIN_NEIGHBORS);
    const fallback=list.filter(x=>x.count<MIN_NEIGHBORS);
    strict.sort((a,b)=>b.score-a.score||b.count-a.count||b.pred-a.pred);
    fallback.sort((a,b)=>b.score-a.score||b.count-a.count||b.pred-a.pred);
    return [...strict,...fallback].slice(0,LIMIT);
}

async function main(){
    const state=await loadJson(STATE_PATH);
    if(!state?.reference){dv.paragraph("Открой карточку фильма и нажми **🔎 Найти похожие**.");return;}
    const refFile=app.vault.getAbstractFileByPath(state.reference);
    if(!refFile||!isMedia(refFile)){dv.paragraph(`Не удалось открыть референс: ${state.reference}`);return;}
    const model=await loadJson(MODEL_PATH);
    if(!model?.users?.length){dv.paragraph("Не найдена MovieLens-модель. Сначала запусти `Кино/_system/Обновить модель MovieLens.bat`.");return;}
    if(!model?.catalog || !Object.keys(model.catalog).length){dv.paragraph("В MovieLens-модели нет каталога названий. Один раз запусти `Кино/_system/Обновить каталог рекомендаций.bat`.");return;}

    const allFiles=app.vault.getMarkdownFiles().filter(isMedia);
    const items=allFiles.map(featureFromFile);
    const matched=assignModelKeys(items,model);
    const ref=items.find(x=>x.file.path===refFile.path);
    const rated=items.filter(x=>x.file.path!==refFile.path&&x.rating!==null&&x.rating>=1&&x.rating<=10);
    const localByKey=new Map(items.filter(x=>x.modelKey).map(x=>[x.modelKey,x]));
    const watchedKeys=new Set(items.filter(x=>x.rating!==null&&x.modelKey).map(x=>x.modelKey));
    const mode=ref.rating!==null&&ref.rating<5?"alternative":ref.rating!==null&&ref.rating>5?"similar":"neutral";
    const modeText=mode==="alternative"?"Альтернативы по той же теме":mode==="similar"?"Похожие, которые должны понравиться":"Нейтральный поиск по теме";

    const head=dv.container.createDiv(); head.createEl("h2",{text:`${modeText}: ${refFile.basename}`});
    const hp=head.createEl("p"); hp.innerHTML=`Твоя оценка: <b>${ref.rating===null?"нет":fmt(ref.rating)}</b> · Прогноз карточки: <b>${fmt(ref.storedPrediction)}</b>`;
    const ui=makeUi(dv.container);
    setProgress(ui,"Шаг 1/2 · считаю MovieLens-рекомендации…",8);
    await new Promise(r=>setTimeout(r,0));

    const cal=calibration(items), globalMean=mean(rated.map(x=>x.rating))||6;
    const agg=neighborAggregate(model,rated,ref,mode);
    let list=[];
    for(const [key,a] of agg){
        if(watchedKeys.has(key)) continue;
        const x=modelCandidate(key,a,model,localByKey,ref,mode,cal,globalMean);
        if(ref.genres.size && x.genres.size && x.theme<0.34) continue;
        if(mode==="alternative"&&ref.rating!==null&&x.pred<Math.max(5.5,ref.rating+0.8)) continue;
        list.push(x);
    }
    list.sort((a,b)=>b.score-a.score||b.count-a.count||b.pred-a.pred);
    list=list.slice(0,CANDIDATE_POOL);
    let top=pickTop(list);

    const cache=(await loadJson(KP_CACHE_PATH))||{};
    // Сначала мгновенно применяем локальные и уже закэшированные КП-ID.
    let found=0;
    for(const x of top){
        if(x.local?.kpId){x.kpId=x.local.kpId;x.ruTitle=x.local.ruTitle;found++;continue;}
        const c=cache[x.modelKey];
        if(c?.found&&c?.kpId){x.kpId=c.kpId;if(c.ruTitle)x.ruTitle=c.ruTitle;if(c.enTitle)x.enTitle=c.enTitle;if(!x.year&&c.year)x.year=c.year;found++;}
    }
    setProgress(ui,`Шаг 1/2 · MovieLens готов · ${top.length} кандидатов · локально сопоставлено с КП: ${found}`,35);
    renderTable(ui,top);
    await new Promise(r=>setTimeout(r,0));

    let dirty=false, done=0, transient=0;
    const unresolved=top.filter(x=>!x.kpId);
    setProgress(ui,unresolved.length?`Шаг 2/2 · Кинопоиск: 0/${unresolved.length} · найдено всего: ${found}`:`Готово · все ${found}/${top.length} уже были в кэше/базе`,unresolved.length?40:100);

    for(const x of unresolved){
        const r=await resolveKinopoisk(x,cache);
        done++;
        if(r?.found){
            if(r.kpId)x.kpId=r.kpId;
            if(r.ruTitle)x.ruTitle=r.ruTitle;
            if(r.enTitle)x.enTitle=r.enTitle;
            if(!x.year&&r.year)x.year=r.year;
            if(r.kpRating!==null&&r.kpRating!==undefined)x.kpRating=r.kpRating;
            found++; dirty=true;
        } else if(r?.transient) transient++;
        else dirty=true;
        const pct=40+Math.round(55*done/Math.max(1,unresolved.length));
        setProgress(ui,`Шаг 2/2 · Кинопоиск: ${done}/${unresolved.length} · найдено всего: ${found}/${top.length}`,pct);
        renderTable(ui,top);
        await new Promise(r=>setTimeout(r,0));
    }
    if(dirty) await saveJson(KP_CACHE_PATH,cache);

    const oneCount=top.filter(x=>x.count===1).length;
    const stale=Math.abs((model.personal_ratings_at_build||rated.length)-rated.length);
    let note=`Готово · Кинопоиск: ${found}/${top.length} · IMDb не используется`;
    if(transient) note+=` · временных ошибок КП: ${transient}`;
    if(oneCount) note+=` · слабая поддержка у ${oneCount} кандид.`;
    if(stale>=30) note+=` · MovieLens-модель старее текущей базы на ~${stale} оценок`;
    note+=` · карточек сопоставлено с MovieLens по названию/году: ${matched}/${items.length}`;
    setProgress(ui,note,100);
}

await main();
```
