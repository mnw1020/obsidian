# Тест источников рекомендаций

Нажми кнопку у сервиса. Obsidian выполнит запрос через `requestUrl`. Для IMDb тест сравнит ответ страницы фильма с отдельным запросом списка «More like this»; для остальных источников покажет найденные фильмы, годы и описания.

```dataviewjs
const http = (typeof requestUrl !== "undefined" && requestUrl) ||
  ((typeof obsidian !== "undefined" && obsidian.requestUrl) ? obsidian.requestUrl : null);

const services = [
  { name: "IMDb", url: "https://www.imdb.com/title/tt0499549/" },
  { name: "MovieTon", url: "https://movieton.org/similar/251733" },
  { name: "LikeFilm", url: "https://likefilm.ru/film/avatar-2009/similar/" },
];
const root = dv.container.createDiv();
root.style.display = "grid";
root.style.gap = "14px";

function decodeHtml(value) {
  return String(value || "").replace(/&nbsp;|&#160;/gi, " ").replace(/&amp;/gi, "&")
    .replace(/&quot;/gi, '"').replace(/&#39;|&apos;/gi, "'")
    .replace(/&lt;/gi, "<").replace(/&gt;/gi, ">");
}
function cleanText(value) {
  return decodeHtml(String(value || "").replace(/<script\b[\s\S]*?<\/script>/gi, " ")
    .replace(/<style\b[\s\S]*?<\/style>/gi, " ").replace(/<[^>]+>/g, " "))
    .replace(/\s+/g, " ").trim();
}
function cardFor(service) {
  const card = root.createDiv();
  card.style.border = "1px solid var(--background-modifier-border)";
  card.style.borderRadius = "8px";
  card.style.padding = "12px";
  card.createEl("h3", { text: service.name });
  const input = card.createEl("input", { type: "text", value: service.url });
  input.style.width = "100%";
  input.style.marginBottom = "8px";
  const controls = card.createDiv();
  controls.style.display = "flex";
  controls.style.gap = "8px";
  const button = controls.createEl("button", { text: "Проверить из Obsidian" });
  const open = controls.createEl("a", { text: "Открыть страницу" });
  open.href = service.url;
  open.target = "_blank";
  const result = card.createEl("pre", { text: "Пока не проверено." });
  result.style.whiteSpace = "pre-wrap";
  result.style.marginTop = "10px";
  input.addEventListener("input", () => { open.href = input.value.trim(); });
  button.addEventListener("click", async () => {
    const url = input.value.trim();
    open.href = url;
    if (!http) { result.textContent = "requestUrl недоступен в этом DataviewJS-контексте."; return; }
    if (!/^https:\/\//i.test(url)) { result.textContent = "Укажи полный URL, начинающийся с https://"; return; }
    button.disabled = true;
    result.textContent = "Запрашиваю страницу…";
    try {
      if (service.name === "IMDb") {
        const referenceId = url.match(/\/title\/(tt\d{7,12})/i)?.[1]?.toLowerCase() || "";
        if (!referenceId) { result.textContent = "В ссылке IMDb не найден ID фильма вида tt1234567."; return; }
        const query = "query Recommendations($id: ID!) { title(id: $id) { moreLikeThisTitles(first: 12) { edges { node { id titleText { text } originalTitleText { text } releaseYear { year } } } } } }";
        const payload = { query, operationName: "Recommendations", variables: { id: referenceId } };
        const endpoint = "https://caching.graphql.imdb.com/";
        const attempts = [
          { name: "IMDb GraphQL GET", request: { url: `${endpoint}?operationName=Recommendations&variables=${encodeURIComponent(JSON.stringify(payload.variables))}&query=${encodeURIComponent(query)}`, method: "GET", throw: false } },
          { name: "IMDb GraphQL POST без дополнительных заголовков", request: { url: endpoint, method: "POST", throw: false, contentType: "application/json", body: JSON.stringify(payload) } },
          { name: "IMDb GraphQL POST с Origin/Referer", request: { url: endpoint, method: "POST", throw: false, contentType: "application/json", headers: { Accept: "application/json", Origin: "https://www.imdb.com", Referer: url }, body: JSON.stringify(payload) } }
        ];
        const lines = [];
        try {
          const page = await http({ url, method: "GET", throw: false, headers: { Accept: "text/html" } });
          lines.push(`Страница фильма: HTTP ${page?.status || 0}, ${String(page?.text || "").length} символов`);
        } catch (error) { lines.push(`Страница фильма: ${String(error?.message || error)}`); }
        for (const attempt of attempts) {
          result.textContent = [...lines, `${attempt.name}: выполняется…`].join("\n");
          try {
            const response = await http(attempt.request);
            const data = response?.json || JSON.parse(String(response?.text || "{}"));
            const films = data?.data?.title?.moreLikeThisTitles?.edges?.map(edge => edge.node).filter(node => /^tt\d{7,12}$/i.test(node?.id || "")) || [];
            lines.push(`${attempt.name}: HTTP ${response?.status || 0}, фильмов ${films.length}`);
            if (data?.errors?.length) lines.push(`Ошибка IMDb: ${data.errors.map(error => error.message).join("; ")}`);
            if (films.length) {
              lines.push(...films.map((film, i) => `${i + 1}. ${film.titleText?.text || film.originalTitleText?.text || film.id} (${film.releaseYear?.year || "год неизвестен"}) — https://www.imdb.com/title/${film.id}/`));
              break;
            }
          } catch (error) { lines.push(`${attempt.name}: ${String(error?.message || error)}`); }
        }
        result.textContent = lines.join("\n");
        return;
      }
      const response = await http({
        url, method: "GET", throw: false,
        headers: { "User-Agent": "Mozilla/5.0", "Accept": "text/html,application/xhtml+xml" }
      });
      const html = String(response?.text || "");
      const status = Number(response?.status || 0);
      const title = html.match(/<title[^>]*>([\s\S]*?)<\/title>/i)?.[1] || "";
      const pageText = cleanText(html);
      // Next.js may stream a React Server Components (Flight) payload instead of rendered HTML.
      const flightChunks = [];
      for (const chunk of html.matchAll(/self\.__next_f\.push\(\[1,"((?:\\.|[^"\\])*)"\]\)/g)) {
        try { flightChunks.push(JSON.parse(`"${chunk[1]}"`)); } catch (_) {}
      }
      const flightText = flightChunks.join("");
      const filmLinks = [];
      for (const match of html.matchAll(/<a\b([^>]*?)href\s*=\s*(["'])(.*?)\2([^>]*)>([\s\S]*?)<\/a>/gi)) {
        const href = decodeHtml(match[3]).trim();
        if (!/(?:\/film\/|\/movie\/|\/title\/|\/similar\/)/i.test(href)) continue;
        const label = cleanText(match[5]);
        if (!label || /похожие фильмы|similar/i.test(label)) continue;
        let absolute = href;
        try { absolute = new URL(href, url).href; } catch (_) {}
        if (!filmLinks.some(x => x.url === absolute)) filmLinks.push({ title: label, url: absolute });
      }
      // MovieTon renders its results as <article role="button"> cards, not <a> links.
      // Parse those cards from the section that labels the similar-movie carousel.
      const movieCards = [];
      const ariaMarker = html.search(/aria-label\s*=\s*(["'])Похожие фильмы\1/i);
      const textMarker = html.search(/Похожие фильмы|Похожие на|Similar movies|Similar films/i);
      const marker = ariaMarker >= 0 ? ariaMarker : textMarker;
      if (marker >= 0) {
        const sectionStart = html.lastIndexOf("<section", marker);
        let similarHtml = html;
        if (sectionStart >= 0) {
          const sectionTag = /<\/?section\b[^>]*>/gi;
          sectionTag.lastIndex = sectionStart;
          let depth = 0, sectionEnd = html.length, tag;
          while ((tag = sectionTag.exec(html))) {
            if (/^<section\b/i.test(tag[0])) depth++;
            else if (--depth === 0) { sectionEnd = sectionTag.lastIndex; break; }
          }
          similarHtml = html.slice(sectionStart, sectionEnd);
        }
        const parseArticleCards = source => {
          for (const match of source.matchAll(/<article\b([^>]*)>([\s\S]*?)<\/article>/gi)) {
            if (!/\brole\s*=\s*(["'])button\1/i.test(match[1])) continue;
            const title = cleanText(match[2].match(/<h3\b[^>]*>([\s\S]*?)<\/h3>/i)?.[1] || "");
            if (!title) continue;
            const cardText = cleanText(match[2]);
            const year = cardText.match(/\b((?:18|19|20)\d{2})\b/)?.[1] || "";
            const ratingMatch = match[2].match(/lucide-star[\s\S]*?<\/svg>\s*<[^>]*>(\d+(?:[.,]\d+)?)/i);
            const id = match[1].match(/\bid\s*=\s*(["'])(.*?)\1/i)?.[2] || "";
            movieCards.push({ title, year, rating: ratingMatch?.[1] || "", id });
          }
        };
        parseArticleCards(similarHtml);
        // Some server responses omit the carousel section wrapper; retry globally.
        if (!movieCards.length && similarHtml !== html) parseArticleCards(html);
      }
      if (!movieCards.length && flightText) {
        const merged = new Map();
        for (const match of flightText.matchAll(/"similars"\s*:\s*\[/g)) {
          const start = match.index + match[0].lastIndexOf("[");
          let depth = 0, inString = false, escaped = false, end = -1;
          for (let i = start; i < flightText.length; i++) {
            const c = flightText[i];
            if (inString) { if (escaped) escaped = false; else if (c === "\\") escaped = true; else if (c === '"') inString = false; continue; }
            if (c === '"') { inString = true; continue; }
            if (c === "[") depth++;
            else if (c === "]" && --depth === 0) { end = i + 1; break; }
          }
          if (end < 0) continue;
          try {
            const list = JSON.parse(flightText.slice(start, end));
            if (!Array.isArray(list)) continue;
            for (const item of list) {
              const id = String(item?.filmId || item?.kinopoiskId || item?.id || "").replace(/\D/g, "");
              if (!id) continue;
              const previous = merged.get(id) || {};
              for (const [key, value] of Object.entries(item)) if (value !== null && value !== undefined && value !== "") previous[key] = value;
              merged.set(id, previous);
            }
          } catch (_) {}
        }
        for (const [id, item] of merged) {
          const title = String(item.nameRu || item.nameEn || item.title || "").trim();
          if (!title) continue;
          movieCards.push({title, year: String(item.yearNumeric || item.year || item.yearText || "").match(/(?:18|19|20)\d{2}/)?.[0] || "",
            rating: String(item.ratingKinopoisk || item.rating || ""), id,
            description: String(item.description || item.shortDescription || item.short_description || item.overview || "").trim()});
        }
      }
      if (service.name === "LikeFilm") {
        movieCards.length = 0;
        filmLinks.length = 0; // Navigation links are not recommendations.
        const starts = [...html.matchAll(/<div\b[^>]*\bid=["']movieSection\d+["'][^>]*class=["'][^"']*uiMovieListSimpleSection[^"']*["'][^>]*>/gi)];
        for (let i = 0; i < starts.length; i++) {
          const card = html.slice(starts[i].index, starts[i+1]?.index || html.length);
          const link = card.match(/<a\b[^>]*href=["']([^"']+)["'][^>]*class=["'][^"']*uiH2[^"']*["'][^>]*>([\s\S]*?)<\/a>/i);
          if (!link) continue;
          const rawTitle = cleanText(link[2]);
          const year = rawTitle.match(/\((?:18|19|20)\d{2}\)\s*$/)?.[0].replace(/\D/g, "") || "";
          const title = rawTitle.replace(/\s*\((?:18|19|20)\d{2}\)\s*$/, "").trim();
          const description = cleanText(card.match(/<p\b[^>]*class=["'][^"']*uiFilmCardDescription[^"']*["'][^>]*>([\s\S]*?)<\/p>/i)?.[1] || "");
          let filmUrl = "";
          try { filmUrl = new URL(link[1], url).href; } catch (_) {}
          if (title && /^https:\/\/likefilm\.ru\/film\//i.test(filmUrl)) movieCards.push({title, year, description, url:filmUrl});
        }
      }
      const hasSimilarMarker = /aria-label\s*=\s*(["'])Похожие фильмы\1/i.test(html) ||
        /похожие фильмы|похожие на|похожие|similar movies|similar films/i.test(pageText);
      const verdict = status < 200 || status >= 300
        ? "Сайт не отдал успешный HTTP-ответ из Obsidian."
        : filmLinks.length || movieCards.length
          ? `Похоже, рекомендации доступны: найдено карточек — ${movieCards.length}, ссылок — ${filmLinks.length}.`
          : hasSimilarMarker
            ? "Страница доступна и содержит маркер похожих фильмов, но карточки не удалось извлечь из HTML."
            : "Страница доступна, но список похожих фильмов в HTML не найден.";
      const lines = [
        `HTTP: ${status || "ошибка/тайм-аут"}`,
        `Заголовок: ${cleanText(title) || "не найден"}`,
        `Размер HTML: ${html.length} символов`,
        `Маркер похожих: ${hasSimilarMarker ? "есть" : "не найден"}`,
        `Карточек-фильмов: ${movieCards.length}`,
        `Ссылок на фильмы: ${filmLinks.length}`,
        verdict,
      ];
      if (movieCards.length) lines.push("", ...movieCards.slice(0, 12).map((x,i) =>
        `${i+1}. ${x.title}${x.year ? ` (${x.year})` : ""}${x.rating ? ` · рейтинг ${x.rating}` : ""}${x.id ? ` · ID ${x.id}` : ""}${x.description ? ` · ${x.description.slice(0,180)}` : " · описание пока не найдено"}${x.url ? ` — ${x.url}` : ""}`));
      if (filmLinks.length) lines.push("", ...filmLinks.slice(0, 12).map((x,i) => `${i+1}. ${x.title} — ${x.url}`));
      if (hasSimilarMarker && !movieCards.length && !filmLinks.length) {
        const positions = [...html.matchAll(/Похожие фильмы|Похожие на|Similar movies|Similar films/gi)].map(x => x.index);
        const flightPositions = [...flightText.matchAll(/Похожие фильмы|Похожие на|Similar movies|Similar films/gi)].map(x => x.index);
        const excerpts = (source, indexes) => indexes.slice(-4).map(pos =>
          cleanText(source.slice(Math.max(0, pos - 180), pos + 1000)).slice(0, 850));
        const knownTitles = ["Кинг Конг", "Первому игроку приготовиться", "Варкрафт", "Принцесса Мононоке"]
          .filter(name => html.includes(name) || flightText.includes(name));
        const titleExcerpts = ["Кинг Конг", "Первому игроку приготовиться", "Варкрафт"]
          .flatMap(name => [...flightText.matchAll(new RegExp(name, "gi"))].slice(0, 2).map(hit =>
            `${name}: ${cleanText(flightText.slice(Math.max(0, hit.index - 500), hit.index + 900)).slice(0, 1100)}`));
        const similarKeys = [...flightText.matchAll(/.{0,250}(?:similars|similarMovies|similarFilms).{0,700}/gi)]
          .slice(0, 4).map(hit => cleanText(hit[0]).slice(0, 900));
        lines.push("", `Диагностика HTML: article=${(html.match(/<article\b/gi) || []).length}, h3=${(html.match(/<h3\b/gi) || []).length}, role=button=${(html.match(/role\s*=\s*(["'])button\1/gi) || []).length}, ссылок=${(html.match(/<a\b/gi) || []).length}.`,
          `Next.js Flight: блоков=${flightChunks.length}, размер=${flightText.length}, маркеров в HTML=${positions.length}, в Flight=${flightPositions.length}.`,
          `Из известных названий рекомендаций в ответе найдены: ${knownTitles.join("; ") || "ни одного"}.`,
          ...similarKeys.map((x,i) => `Ключ similars ${i+1}: ${x}`),
          ...titleExcerpts.map((x,i) => `Контекст названия ${i+1}: ${x}`),
          ...excerpts(flightText, flightPositions).map((x,i) => `Контекст Flight ${i+1}: ${x}`),
          ...excerpts(html, positions).map((x,i) => `Контекст HTML ${i+1}: ${x}`));
      }
      result.textContent = lines.join("\n");
    } catch (error) {
      result.textContent = `Запрос не выполнен: ${String(error?.message || error)}`;
    } finally { button.disabled = false; }
  });
}
for (const service of services) cardFor(service);
```
