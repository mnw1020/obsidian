# Тест источников рекомендаций

Тестовый фильм: **Пассажиры (2016)**. Кнопка отправляет GET-запрос из Obsidian через `requestUrl` и проверяет ответ страницы. Я Кино получает конкретную страницу фильма с похожими. КиноКомпас — персональная подборка из профиля, не похожие к одному фильму. Для Nextfilm и Кинонавигатора вставь в поле URL страницу «Пассажиров», если найдёшь её на сайте. Запрос `requestUrl` не использует авторизацию из отдельного браузера, поэтому для КиноКомпаса может показать страницу входа вместо персональных рекомендаций.

```dataviewjs
const http = (typeof requestUrl !== "undefined" && requestUrl) ||
  ((typeof obsidian !== "undefined" && obsidian.requestUrl) ? obsidian.requestUrl : null);

const services = [
  { name: "Я Кино", url: "https://yakino.ru/passengers-2016/", note: "Известная страница Пассажиров с подборкой похожих фильмов." },
  { name: "КиноКомпас", url: "https://kinocensor.ru/cabinet/recommendations", note: "Персональные рекомендации профиля (не похожие к одному фильму). Для теста авторизуйся на сайте; requestUrl может не передавать вход из браузера." },
  { name: "Nextfilm", url: "https://nextfilm.pro/", note: "Найди Пассажиров на сайте и вставь адрес страницы фильма в поле." },
  { name: "Кинонавигатор", url: "https://kinonavigator.ru/", note: "Найди Пассажиров на сайте и вставь адрес страницы фильма в поле." },
];
const root = dv.container.createDiv();
root.style.display = "grid";
root.style.gap = "14px";
root.createEl("p", { text: "Проверка выполняется только после нажатия кнопки. HTTP 200 означает, что Obsidian получил страницу; наличие рекомендаций оценивается отдельно по содержимому ответа." });

function cleanText(html) {
  return String(html || "").replace(/<script\b[\s\S]*?<\/script>/gi, " ")
    .replace(/<style\b[\s\S]*?<\/style>/gi, " ").replace(/<[^>]+>/g, " ")
    .replace(/&nbsp;|&#160;/gi, " ").replace(/&amp;/gi, "&")
    .replace(/&quot;/gi, '"').replace(/&#39;|&apos;/gi, "'")
    .replace(/\s+/g, " ").trim();
}
function addService(service) {
  const card = root.createDiv();
  card.style.border = "1px solid var(--background-modifier-border)";
  card.style.borderRadius = "8px";
  card.style.padding = "12px";
  card.createEl("h3", { text: service.name });
  card.createEl("p", { text: service.note });
  const input = card.createEl("input", { type: "text", value: service.url });
  input.style.width = "100%";
  input.style.marginBottom = "8px";
  const controls = card.createDiv();
  controls.style.display = "flex";
  controls.style.gap = "8px";
  const check = controls.createEl("button", { text: "Проверить из Obsidian" });
  const open = controls.createEl("a", { text: "Открыть сайт" });
  open.href = service.url;
  open.target = "_blank";
  const result = card.createEl("pre", { text: "Пока не проверено." });
  result.style.whiteSpace = "pre-wrap";
  result.style.marginTop = "10px";
  input.addEventListener("input", () => { open.href = input.value.trim(); });
  check.addEventListener("click", async () => {
    const url = input.value.trim();
    open.href = url;
    if (!http) { result.textContent = "requestUrl недоступен в этом DataviewJS-контексте."; return; }
    if (!/^https:\/\//i.test(url)) { result.textContent = "Укажи полный адрес, начинающийся с https://"; return; }
    check.disabled = true;
    result.textContent = "Запрашиваю страницу…";
    try {
      const response = await http({
        url, method: "GET", throw: false,
        headers: { "User-Agent": "Mozilla/5.0", "Accept": "text/html,application/xhtml+xml" }
      });
      const html = String(response?.text || "");
      const title = html.match(/<title[^>]*>([\s\S]*?)<\/title>/i)?.[1];
      const text = cleanText(html);
      const recommendationWords = [...new Set((text.match(/похожие фильмы|похожие на|рекомендац\w*|similar movies|more like this/gi) || []).map(x => x.toLocaleLowerCase("ru")))];
      const passengerMention = /пассажиры|passengers/i.test(text);
      const candidateLinks = (html.match(/<a\b[^>]*href=/gi) || []).length;
      const status = Number(response?.status || 0);
      let verdict = status >= 200 && status < 300
        ? (recommendationWords.length ? "Страница доступна; в HTML есть слова о рекомендациях. Проверь глазами, что там именно список фильмов." : "Страница доступна, но явный блок рекомендаций в HTML не найден. Возможно, он загружается JavaScript-ом, требует входа или URL ведёт не на карточку фильма.")
        : "Страница не отдала успешный HTTP-ответ из Obsidian.";
      if (status >= 200 && status < 300 && !passengerMention && /yakino|kinocensor/i.test(url)) verdict += " В HTML не найдено упоминание Пассажиров — проверь адрес страницы.";
      result.textContent = [
        `HTTP: ${status || "ошибка/тайм-аут"}`,
        `Заголовок: ${cleanText(title || "") || "не найден"}`,
        `Размер HTML: ${html.length} символов · ссылок: ${candidateLinks}`,
        `Упоминание Пассажиров: ${passengerMention ? "да" : "нет"}`,
        `Маркеры рекомендаций: ${recommendationWords.join(", ") || "не найдены"}`,
        verdict
      ].join("\n");
    } catch (error) {
      result.textContent = `Запрос не выполнен: ${String(error?.message || error)}`;
    } finally { check.disabled = false; }
  });
}
for (const service of services) addService(service);
```
