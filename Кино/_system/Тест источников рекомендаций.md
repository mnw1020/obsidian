# Тест источников рекомендаций

Нажми кнопку у сервиса. Obsidian выполнит запрос через `requestUrl`, покажет ответ страницы и попробует извлечь ссылки на фильмы из списка похожих.

```dataviewjs
const http = (typeof requestUrl !== "undefined" && requestUrl) ||
  ((typeof obsidian !== "undefined" && obsidian.requestUrl) ? obsidian.requestUrl : null);

const services = [
  { name: "MovieTon", url: "https://movieton.org/similar/251733" },
  { name: "LikeFilm", url: "https://likefilm.ru/film/avatar-2009/similar/" },
];
const root = dv.container.createDiv();
root.style.display = "grid";
root.style.gap = "14px";

function decodeHtml(value) {
  return String(value || "").replace(/&nbsp;|&#160;/gi, " ").replace(/&amp;/gi, "&")
    .replace(/&quot;/gi, '"').replace(/&#39;|&apos;/gi, "'")
    .replace(/&lt;/gi, "<").replace(/&gt;/gi, ">");
}
function cleanText(value) {
  return decodeHtml(String(value || "").replace(/<script\b[\s\S]*?<\/script>/gi, " ")
    .replace(/<style\b[\s\S]*?<\/style>/gi, " ").replace(/<[^>]+>/g, " "))
    .replace(/\s+/g, " ").trim();
}
function cardFor(service) {
  const card = root.createDiv();
  card.style.border = "1px solid var(--background-modifier-border)";
  card.style.borderRadius = "8px";
  card.style.padding = "12px";
  card.createEl("h3", { text: service.name });
  const input = card.createEl("input", { type: "text", value: service.url });
  input.style.width = "100%";
  input.style.marginBottom = "8px";
  const controls = card.createDiv();
  controls.style.display = "flex";
  controls.style.gap = "8px";
  const button = controls.createEl("button", { text: "Проверить из Obsidian" });
  const open = controls.createEl("a", { text: "Открыть страницу" });
  open.href = service.url;
  open.target = "_blank";
  const result = card.createEl("pre", { text: "Пока не проверено." });
  result.style.whiteSpace = "pre-wrap";
  result.style.marginTop = "10px";
  input.addEventListener("input", () => { open.href = input.value.trim(); });
  button.addEventListener("click", async () => {
    const url = input.value.trim();
    open.href = url;
    if (!http) { result.textContent = "requestUrl недоступен в этом DataviewJS-контексте."; return; }
    if (!/^https:\/\//i.test(url)) { result.textContent = "Укажи полный URL, начинающийся с https://"; return; }
    button.disabled = true;
    result.textContent = "Запрашиваю страницу…";
    try {
      const response = await http({
        url, method: "GET", throw: false,
        headers: { "User-Agent": "Mozilla/5.0", "Accept": "text/html,application/xhtml+xml" }
      });
      const html = String(response?.text || "");
      const title = html.match(/<title[^>]*>([\s\S]*?)<\/title>/i)?.[1] || "";
      const pageText = cleanText(html);
      const filmLinks = [];
      for (const match of html.matchAll(/<a\b([^>]*?)href\s*=\s*(["'])(.*?)\2([^>]*)>([\s\S]*?)<\/a>/gi)) {
        const href = decodeHtml(match[3]).trim();
        if (!/(?:\/film\/|\/movie\/|\/title\/|\/similar\/)/i.test(href)) continue;
        const label = cleanText(match[5]);
        if (!label || /похожие фильмы|similar/i.test(label)) continue;
        let absolute = href;
        try { absolute = new URL(href, url).href; } catch (_) {}
        if (!filmLinks.some(x => x.url === absolute)) filmLinks.push({ title: label, url: absolute });
      }
      // MovieTon renders its results as <article role="button"> cards, not <a> links.
      // Parse those cards from the section that labels the similar-movie carousel.
      const movieCards = [];
      const marker = html.search(/aria-label\s*=\s*(["'])Похожие фильмы\1/i);
      if (marker >= 0) {
        const sectionStart = html.lastIndexOf("<section", marker);
        if (sectionStart >= 0) {
          const sectionTag = /<\/?section\b[^>]*>/gi;
          sectionTag.lastIndex = sectionStart;
          let depth = 0, sectionEnd = html.length, tag;
          while ((tag = sectionTag.exec(html))) {
            if (/^<section\b/i.test(tag[0])) depth++;
            else if (--depth === 0) { sectionEnd = sectionTag.lastIndex; break; }
          }
          const similarHtml = html.slice(sectionStart, sectionEnd);
          for (const match of similarHtml.matchAll(/<article\b([^>]*)>([\s\S]*?)<\/article>/gi)) {
            if (!/\brole\s*=\s*(["'])button\1/i.test(match[1])) continue;
            const title = cleanText(match[2].match(/<h3\b[^>]*>([\s\S]*?)<\/h3>/i)?.[1] || "");
            if (!title) continue;
            const cardText = cleanText(match[2]);
            const year = cardText.match(/\b((?:18|19|20)\d{2})\b/)?.[1] || "";
            const ratingMatch = match[2].match(/lucide-star[\s\S]*?<\/svg>\s*<[^>]*>(\d+(?:[.,]\d+)?)/i);
            const id = match[1].match(/\bid\s*=\s*(["'])(.*?)\1/i)?.[2] || "";
            movieCards.push({ title, year, rating: ratingMatch?.[1] || "", id });
          }
        }
      }
      const hasSimilarMarker = /aria-label\s*=\s*(["'])Похожие фильмы\1/i.test(html) ||
        /похожие фильмы|похожие на|похожие|similar movies|similar films/i.test(pageText);
      const status = Number(response?.status || 0);
      const verdict = status < 200 || status >= 300
        ? "Сайт не отдал успешный HTTP-ответ из Obsidian."
        : filmLinks.length || movieCards.length
          ? `Похоже, рекомендации доступны: найдено карточек — ${movieCards.length}, ссылок — ${filmLinks.length}.`
          : hasSimilarMarker
            ? "Страница доступна и содержит маркер похожих фильмов, но карточки не удалось извлечь из HTML."
            : "Страница доступна, но список похожих фильмов в HTML не найден.";
      const lines = [
        `HTTP: ${status || "ошибка/тайм-аут"}`,
        `Заголовок: ${cleanText(title) || "не найден"}`,
        `Размер HTML: ${html.length} символов`,
        `Маркер похожих: ${hasSimilarMarker ? "есть" : "не найден"}`,
        `Карточек-фильмов: ${movieCards.length}`,
        `Ссылок на фильмы: ${filmLinks.length}`,
        verdict,
      ];
      if (movieCards.length) lines.push("", ...movieCards.slice(0, 12).map((x,i) =>
        `${i+1}. ${x.title}${x.year ? ` (${x.year})` : ""}${x.rating ? ` · рейтинг ${x.rating}` : ""}${x.id ? ` · ID ${x.id}` : ""}`));
      if (filmLinks.length) lines.push("", ...filmLinks.slice(0, 12).map((x,i) => `${i+1}. ${x.title} — ${x.url}`));
      result.textContent = lines.join("\n");
    } catch (error) {
      result.textContent = `Запрос не выполнен: ${String(error?.message || error)}`;
    } finally { button.disabled = false; }
  });
}
for (const service of services) cardFor(service);
```
