# Проверка локальной модели

[[Кино/_index|← Кинотека]] · [[Кино/_system/Прогноз/README|О прогнозах]]

Кнопка загрузит **EmbeddingGemma**, вычислит два новых вектора и выгрузит модель. Проверяются размерность, конечные значения и нормировка; кеш кинотеки не используется. Чтобы проверить автономную работу, сначала полностью синхронизируй служебную папку модели, затем отключи Интернет на этом устройстве и нажми кнопку. Результат Android записывается только при запуске в настоящем Android-приложении Obsidian. Первая проверка модели может занять несколько минут.

```dataviewjs
const doc = dv.container.ownerDocument;
const obsidian = typeof require === "function" ? require("obsidian") : {};
const platform = obsidian.Platform?.isAndroidApp === true ? "android"
    : obsidian.Platform?.isDesktopApp === true ? "desktop" : "unknown";
const deviceLabel = platform === "android" ? "Android" : platform === "desktop" ? "ПК" : "это устройство";
const diagnosticPath = "Кино/_system/Прогноз/device-check-" + (platform === "android" ? "android" : "desktop") + ".json";
const root = doc.createElement("section"); root.className = "kino-synopsis";
dv.container.appendChild(root);
const button = doc.createElement("button"); button.type = "button";
button.className = "kino-action"; button.textContent = "Проверить на этом устройстве";
root.appendChild(button);
const cancel = doc.createElement("button"); cancel.type = "button"; cancel.className = "kino-action";
cancel.textContent = "Отменить проверку"; cancel.hidden = true; cancel.style.display = "none"; root.appendChild(cancel);
const status = doc.createElement("p"); status.className = "kino-forecast-muted";
status.setAttribute("role", "status"); status.setAttribute("aria-live", "polite");
status.textContent = "Проверка ещё не запускалась на этом устройстве."; root.appendChild(status);
let disposed = false, running = false, runtime = null, controller = null;
const stop = () => { controller?.abort(); runtime?.dispose(); runtime = null; };
cancel.addEventListener("click", stop);
dv.component?.register?.(() => { disposed = true; stop(); root.remove(); });
function safeError(error) {
    const message = String(error?.message ?? "");
    if (error?.name === "AbortError") return "Проверка прервана. Нажми кнопку, чтобы повторить её.";
    if (/vector|вектор|размерност|нормировк|число результатов/i.test(message)) return "Модель вернула некорректные векторы: проверка размерности, чисел или нормировки не пройдена.";
    if (/memory|alloc|памят/i.test(message)) return "Недостаточно памяти для локальной модели. Закрой лишние приложения и повтори проверку.";
    if (/version|верси/i.test(message)) return "Версии файлов модели и библиотек не совпадают. Полностью синхронизируй служебную папку модели.";
    if (/ENOENT|not found|не найден|отсутств|синхрониз|поврежд/i.test(message)) return "Не все файлы локальной модели доступны. Полностью синхронизируй папку «Кино/_system/Прогноз/runtime» и повтори проверку.";
    if (/timeout|вовремя/i.test(message)) return "Модель не ответила вовремя. Оставь Obsidian открытым, освободи память и повтори проверку.";
    if (/Worker|WebAssembly|wasm|SIMD/i.test(message)) return "Не удалось запустить локальную модель. Техническая причина сохранена в диагностике.";
    return "Локальная модель не запустилась. Проверь синхронизацию её файлов, свободную память и версию Obsidian.";
}
function technicalError(error) {
    return { name: String(error?.name || "Error"), message: String(error?.message || error).slice(0, 4000), stack: String(error?.stack || "").slice(0, 6000) };
}
const previous = app.vault.getAbstractFileByPath(diagnosticPath);
if (previous) {
    try {
        const saved = JSON.parse(await app.vault.read(previous));
        if (!disposed && saved.platform === platform && saved.version === 4 && typeof saved.passed === "boolean") {
            const date = new Date(saved.date);
            status.textContent = "Последняя проверка (" + deviceLabel + "): " + (saved.passed ? "пройдена" : "не пройдена")
                + (Number.isNaN(date.getTime()) ? "" : " · " + date.toLocaleString("ru-RU")) + ". Кнопка выполнит новый расчёт.";
        }
    } catch (_) { /* A missing or older diagnostic does not prevent a fresh check. */ }
}
button.addEventListener("click", async () => {
    if (running || disposed) return;
    if (app.__kinoRatingUpdate || app.__kinoForecastResuming || app.__kinoForecastBusy) {
        status.textContent = "Сейчас обновляются прогнозы. Дождись окончания обновления и повтори проверку.";
        return;
    }
    running = true; button.disabled = true; button.setAttribute("aria-busy", "true"); cancel.hidden = false; cancel.style.display = "";
    controller = new AbortController();
    const started = performance.now();
    const diagnostic = { version: 4, date: new Date().toISOString(), platform, passed: false, elapsedMs: 0, models: {}, error: null, cancelled: false,
        environment: { userAgent: navigator.userAgent, origin: location.origin, protocol: location.protocol, worker: typeof Worker, webAssembly: typeof WebAssembly } };
    try {
        status.textContent = "Загружается локальная модель на «" + deviceLabel + "». Оставь Obsidian открытым…";
        const file = app.vault.getAbstractFileByPath("Кино/_system/semantic_runtime.js");
        if (!file) throw new Error("Не найден semantic_runtime.js");
        const mod = { exports: {} };
        new Function("module", await app.vault.read(file))(mod);
        const keys = ["embeddinggemma"];
        const models = mod.exports.MODELS;
        if (!Array.isArray(models) || keys.some(key => !models.some(model => model.key === key))) throw new Error("Версии файлов модели не совпадают");
        if (disposed) return;
        for (const modelKey of keys) {
            if (disposed || controller.signal.aborted) break;
            const model = models.find(value => value.key === modelKey), modelStarted = performance.now();
            const result = { label: model.label, revision: model.revision, dimensions: model.dimensions, passed: false, elapsedMs: 0, norms: [], error: null };
            diagnostic.models[modelKey] = result;
            try {
                status.textContent = "Загружается " + model.label + " на «" + deviceLabel + "». Оставь Obsidian открытым…";
                runtime = await mod.exports.create({ app, modelKey, signal: controller.signal, idleMs: 0,
                    onProgress(progress) {
                        result.lastProgress = { phase: progress.phase, status: progress.status, file: progress.file };
                        if (!disposed && progress.phase === "embedding") status.textContent = model.label + ": свежий расчёт сюжетов " + progress.completed + " / " + progress.total + ".";
                    }
                });
                if (disposed) return;
                const results = await runtime.embed([
                    "Обычный человек не чувствует боли и пытается спасти похищенную девушку, преследуя опасных преступников.",
                    "Молодая исследовательница отправляется в космос и находит заброшенную станцию с загадочным сигналом."
                ]);
                if (disposed) return;
                if (!Array.isArray(results) || results.length !== 2) throw new Error("Неверное число результатов");
                for (const item of results) {
                    const vector = item?.vector;
                    if (!Array.isArray(vector) && !ArrayBuffer.isView(vector)) throw new Error("Некорректный вектор");
                    if (vector.length !== model.dimensions || !Array.from(vector).every(Number.isFinite)) throw new Error("Некорректная размерность вектора");
                    const norm = Math.sqrt(Array.from(vector).reduce((sum, value) => sum + value * value, 0));
                    if (Math.abs(norm - 1) > 0.001) throw new Error("Нормировка вектора не пройдена");
                    result.norms.push(norm);
                }
                result.passed = true;
            } catch (error) {
                result.error = safeError(error);
                result.technicalError = technicalError(error);
                if (error?.name === "AbortError" || controller.signal.aborted) { diagnostic.cancelled = true; break; }
            } finally {
                result.elapsedMs = Math.round(performance.now() - modelStarted);
                runtime?.dispose(); runtime = null;
            }
        }
        diagnostic.cancelled ||= controller.signal.aborted;
        diagnostic.passed = !diagnostic.cancelled && keys.every(key => diagnostic.models[key]?.passed === true);
    } catch (error) {
        diagnostic.error = safeError(error);
        diagnostic.technicalError = technicalError(error);
    } finally {
        diagnostic.elapsedMs = Math.round(performance.now() - started);
        stop(); controller = null;
        if (!disposed) {
            const results = Object.values(diagnostic.models).map(result => result.label + ": "
                + (result.passed ? "пройдена, " + result.dimensions + " чисел, " + (result.elapsedMs / 1000).toFixed(1) + " с" : result.error));
            status.textContent = (diagnostic.passed ? "Проверка пройдена" : diagnostic.cancelled ? "Проверка прервана" : "Проверка не пройдена")
                + " · " + deviceLabel + " · " + (diagnostic.elapsedMs / 1000).toFixed(1) + " с. " + results.join("; ")
                + (diagnostic.error ? " " + diagnostic.error : "");
            if (!status.textContent.endsWith(".")) status.textContent += ".";
            try {
                const body = JSON.stringify(diagnostic, null, 2) + "\n";
                const existing = app.vault.getAbstractFileByPath(diagnosticPath);
                if (existing) await app.vault.modify(existing, body);
                else await app.vault.create(diagnosticPath, body);
                if (!disposed) status.textContent += " Диагностика сохранена.";
            } catch (_) {
                if (!disposed) status.textContent += " Не удалось сохранить диагностику в служебной папке; сам результат проверки показан выше.";
            }
            button.disabled = false; button.removeAttribute("aria-busy"); cancel.hidden = true; cancel.style.display = "none";
        }
        running = false;
    }
});
```

Диагностика после проверки: [[Кино/_system/Прогноз/device-check-android.json|Android]] · [[Кино/_system/Прогноз/device-check-desktop.json|ПК / другое устройство]]. Успешная проверка на ПК подтверждает работу только на ПК. Для проверки Android открой эту заметку и нажми кнопку именно на телефоне; также проверь создание новой карточки и возвращение в Obsidian после сворачивания.
