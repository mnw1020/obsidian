```dataviewjs
const selected = String(dv.current()["Выбрано"] || "");
if (!selected) {
    dv.paragraph("Выбери имя в кинотеке или запусти соответствующую команду QuickAdd.");
} else {
    dv.header(2, selected);
    const rows = dv.pages('"Кино"').array().filter(p => {
        if (/^Кино\/(Просмотры|Сезоны|Франшизы|Служебное|_system)\//.test(p.file.path)) return false;
        const tags = Array.isArray(p.tags) ? p.tags : [p.tags];
        if (!tags.some(t => ["movies", "serial"].includes(String(t).replace(/^#/, "")))) return false;
        const values = Array.isArray(p["Актеры"]) ? p["Актеры"] : [p["Актеры"]];
        return values.includes(selected);
    });
    const ratings = rows.map(p => p["Оценка"]).filter(v => v != null && String(v).trim() !== "")
        .map(v => Number(String(v).replace(",", "."))).filter(v => Number.isFinite(v) && v >= 1 && v <= 10);
    dv.paragraph("Произведений: " + rows.length + " · Средняя моя оценка: " +
        (ratings.length ? (ratings.reduce((a,b) => a+b, 0) / ratings.length).toFixed(2) : "нет оценок"));
}
```
