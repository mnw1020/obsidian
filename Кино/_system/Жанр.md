function kinoText(value) { return String(value ?? "").trim().normalize("NFC"); }
function kinoKey(value) { return kinoText(value).toLocaleLowerCase("ru").replace(/ё/g, "е"); }
function kinoGenre(value) {
    const text = kinoText(value);
    const key = kinoKey(text);
    for (const [canonical, aliases] of Object.entries(KINO_GENRE_ALIASES)) {
        if ([canonical, ...aliases].some(alias => kinoKey(alias) === key)) return canonical;
    }
    return text;
}
