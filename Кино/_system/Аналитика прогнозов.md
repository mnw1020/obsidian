try {
    const file = app.vault.getAbstractFileByPath("Кино/_system/forecast_analytics.js");
    if (!file) throw new Error("Не найден модуль аналитики прогнозов");
    const mod = { exports: {} };
    new Function("module", await app.vault.read(file))(mod);
    await mod.exports({ dv, app, obsidian: typeof require === "function" ? require("obsidian") : {} });

    dv.paragraph("Не удалось загрузить аналитику прогнозов: " + (error.message || error));

}
