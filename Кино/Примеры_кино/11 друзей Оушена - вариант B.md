**Вариант B: ссылки под свойствами.** Нажми на имя в блоке ниже. Отдельный плагин Kino Entity Demo не нужен; используются уже имеющиеся Dataview и QuickAdd.

Оригинал: [[Кино/11 друзей Оушена]]

```dataviewjs
const current = dv.current();
const fields = [
    ["Режисер", "Режиссер", "Кино - Открыть режиссера"],
    ["Актеры", "Актеры", "Кино - Открыть актера"],
    ["Жанр", "Жанры", "Кино - Открыть жанр"]
];
for (const [field, label, choice] of fields) {
    const raw = current[field];
    const values = (Array.isArray(raw) ? raw : [raw])
        .filter(value => typeof value === "string" && value.trim());
    const row = dv.container.createDiv();
    row.createEl("strong", { text: label + ": " });
    values.forEach((name, index) => {
        if (index) row.appendText(" · ");
        const uri = "obsidian://quickadd?vault=" + encodeURIComponent(app.vault.getName())
            + "&choice=" + encodeURIComponent(choice)
            + "&value-entity=" + encodeURIComponent(name);
        row.createEl("a", { text: name, href: uri });
    });
    if (!values.length) row.appendText("Не указано");
}
```
